准备数据：标签数组和 jsonb 属性。
CREATE TABLE products (id serial PRIMARY KEY, name text, tags text[], attrs jsonb);
INSERT INTO products (name, tags, attrs)
SELECT 'p' || i, ARRAY['t' || (i % 50), 't' || (i % 7)],
       jsonb_build_object('color', (ARRAY['red','blue','green'])[1 + i % 3], 'size', i % 5)
FROM generate_series(1, 200000) i;
ANALYZE products;
INSERT 0 200000

02
没有索引：数组包含查询是顺序扫描。
EXPLAIN SELECT count(*) FROM products WHERE tags @> ARRAY['t13'];
   ->  Seq Scan on products
         Filter: (tags @> '{t13}'::text[])
03
GIN 索引。
CREATE INDEX products_tags_gin ON products USING gin (tags);
EXPLAIN SELECT count(*) FROM products WHERE tags @> ARRAY['t13'];
   ->  Bitmap Heap Scan on products
         Recheck Cond: (tags @> '{t13}'::text[])
         ->  Bitmap Index Scan on products_tags_gin
04
jsonb：jsonb_path_ops 版本更小。
CREATE INDEX products_attrs_gin ON products USING gin (attrs);
CREATE INDEX products_attrs_path ON products USING gin (attrs jsonb_path_ops);
SELECT relname, pg_size_pretty(pg_relation_size(oid)) FROM pg_class WHERE relname LIKE 'products_attrs%';
       relname       | pg_size_pretty
---------------------+----------------
 products_attrs_gin  | 1720 kB
 products_attrs_path | 880 kB
05
BRIN：按时间追加的大表。
CREATE TABLE logs AS SELECT g AS id, timestamptz '2026-01-01' + g * interval '1 second' AS ts, md5(g::text) AS msg FROM generate_series(1, 5000000) g;
CREATE INDEX logs_ts_brin ON logs USING brin (ts);
CREATE INDEX logs_ts_btree ON logs (ts);
SELECT relname, pg_size_pretty(pg_relation_size(oid)) FROM pg_class WHERE relname LIKE 'logs_ts%';
    relname    | pg_size_pretty
---------------+----------------
 logs_ts_brin  | 24 kB
 logs_ts_btree | 107 MB
06
表达式索引与部分索引。
CREATE INDEX users_lower_email ON users (lower(email));
EXPLAIN SELECT * FROM users WHERE lower(email) = 'a@x.com';
CREATE INDEX orders_pending ON orders (created_at) WHERE status = 'pending';
EXPLAIN SELECT * FROM orders WHERE status = 'pending' AND created_at < now() - interval '1 hour';
 Index Scan using users_lower_email on users
   Index Cond: (lower(email) = 'a@x.com'::text)
 Bitmap Heap Scan on orders
   ->  Bitmap Index Scan on orders_pending
挑战
用约束防止预订冲突
会议室预订表 bookings(room_id int, during tstzrange)。要求：同一个会议室的时间段不能重叠，由数据库保证，而不是应用先查后插。写出建表语句，并演示冲突时的报错。
查看参考答案
CREATE EXTENSION btree_gist;
CREATE TABLE bookings (
  room_id int,
  during tstzrange,
  EXCLUDE USING gist (room_id WITH =, during WITH &&)
);
INSERT INTO bookings VALUES (1, '[2026-10-22 10:00, 2026-10-22 11:00)');
INSERT INTO bookings VALUES (1, '[2026-10-22 10:30, 2026-10-22 12:00)');
INSERT 0 1
ERROR:  conflicting key value violates exclusion constraint "bookings_room_id_during_excl"
DETAIL:  Key (room_id, during)=(1, ["2026-10-22 10:30:00+08","2026-10-22 12:00:00+08")) conflicts with existing key (room_id, during)=(1, ["2026-10-22 10:00:00+08","2026-10-22 11:00:00+08")).
排他约束是唯一约束的推广：任意两行在所有列上的指定运算符都返回 true 时就冲突。&&（重叠）需要 GiST 索引，room_id 的 = 在 GiST 中由 btree_gist 扩展提供。右开区间 [10:00, 11:00) 让 11:00 开始的下一场不算冲突。先查后插在并发下会漏掉冲突，约束则在索引层面加锁检查。PostgreSQL 18 还支持 PRIMARY KEY (room_id, during WITHOUT OVERLAPS) 的写法，本质相同。

小结
记住这几点


  1. GIN（倒排索引）：把一个值拆成多个元素（数组元素、jsonb 键值、全文词元），为每个元素存一个指向行的列表。适合 @>、?、&&、@@。查询快，写入慢（用 fastupdate 的待处理列表缓冲），索引较大。jsonb 用 jsonb_path_ops 更小，但只支持 @>。
  2. GiST：平衡树框架，每个节点存子树的“外包”（如边界框、范围）。适合几何、范围类型（tstzrange 的 &&）、最近邻 ORDER BY 坐标 <-> 点、排他约束 EXCLUDE USING gist，以及 pg_trgm 的模糊匹配。
  3. BRIN：每一段连续的堆页（默认 128 页）只存最小值和最大值，索引极小（几十 KB 对应几十 GB 的表）。只在列值与物理顺序强相关时有效，典型是只追加的日志表的时间列。
  4. Hash 只支持 =，现在已经写 WAL、可以安全使用，但通常不比 B-tree 有明显优势。SP-GiST 用于四叉树、前缀树等不平衡分区结构（IP 地址、电话前缀）。
  5. 表达式索引：CREATE INDEX ON users (lower(email))，查询必须写成完全相同的表达式才能用上。表达式必须是 IMMUTABLE 函数。
  6. 部分索引：CREATE INDEX ON orders (created_at) WHERE status = 'pending' 只索引一小部分行，更小、更新更少。查询的 WHERE 条件必须能推导出索引条件。部分唯一索引能表达“每个用户最多一个未删除的账号”这类约束。
