01
创建用户表和订单表。
CREATE TABLE users (
  id         bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  email      text NOT NULL UNIQUE CHECK (email LIKE '%@%'),
  name       text NOT NULL CHECK (length(name) <= 100),
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE orders (
  id       bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  user_id  bigint NOT NULL REFERENCES users(id),
  amount   numeric(12,2) NOT NULL CHECK (amount > 0),
  status   text NOT NULL DEFAULT 'pending' CHECK (status IN ('pending','paid','shipped','cancelled')),
  meta     jsonb NOT NULL DEFAULT '{}',
  created_at timestamptz NOT NULL DEFAULT now()
);

挑战
给已有表加约束
orders 表已有大量数据，现在要求 meta 中必须有 "channel" 字段。直接 ALTER TABLE … ADD CONSTRAINT … CHECK 会怎样？写出一种不长时间阻塞写入的做法，并说明如果已有旧数据不满足条件该怎么办。
查看参考答案
ALTER TABLE orders ADD CONSTRAINT orders_meta_channel
  CHECK (meta ? 'channel') NOT VALID;
-- 新写入立即受检查；旧数据先修复
UPDATE orders SET meta = meta || '{"channel":"unknown"}' WHERE NOT meta ? 'channel';
ALTER TABLE orders VALIDATE CONSTRAINT orders_meta_channel;
-- 本挑战验证完毕后，恢复后续课程使用的表结构
ALTER TABLE orders DROP CONSTRAINT orders_meta_channel;
ALTER TABLE
UPDATE 1
ALTER TABLE
直接 ADD CONSTRAINT 会在持有 ACCESS EXCLUSIVE 锁的情况下扫描全表验证，大表上会长时间阻塞所有读写；有一行不满足就整个失败。NOT VALID 只需短暂加锁，从此对新写入和更新生效；VALIDATE CONSTRAINT 只持有较弱的 SHARE UPDATE EXCLUSIVE 锁，扫描期间读写都可以继续。旧数据不满足时先分批修复（大表的 UPDATE 也要分批，见第 25 课）。外键约束同样支持这个两步做法。最后一条 DROP 仅用于恢复课程练习环境：后续章节仍允许 meta 省略 channel；实际业务若需要此约束，应保留它并更新所有写入方。

小结
记住这几点


  1. 主键用 bigint GENERATED ALWAYS AS IDENTITY（或 uuid，PostgreSQL 18 有 uuidv7()，按时间有序，对索引友好）。int 的上限约 21 亿，迟早不够用。
  2. 金额用 numeric(12,2) 或以“分”为单位的 bigint，不要用 float：二进制浮点数无法精确表示 0.1。
  3. 字符串用 text；需要长度上限时加 CHECK 约束，而不是 varchar(n)。在 PostgreSQL 中它们的存储和性能没有区别。
  4. 时间用 timestamptz：存储的是绝对时刻（内部为 UTC），显示时按会话的 TimeZone 转换。timestamp（不带时区）只是一个“墙上时间”，跨时区会出错。
  5. jsonb 适合真正结构不固定的数据；经常查询、需要约束的字段应该是独立的列。
  6. 约束是最后一道防线：应用代码有 bug、有多个写入方、有人手动改数据时，只有数据库约束始终生效。违反约束时返回 SQLSTATE 错误码（23502 NOT NULL、23505 唯一、23503 外键、23514 CHECK），应用可以据此处理。
