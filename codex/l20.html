Codex CLI · 第 20 课

/goal：长时间运行的目标

普通 agent 回合在模型停止调用工具时结束，然后等待用户。/goal 改变了这一点：线程有了一个持久化的目标（objective、状态、token 预算），每当线程空闲，Goals 扩展就自动开启一个新回合，注入“继续推进目标”的提示，直到模型用 update_goal 把目标标记为 complete（或 blocked、被用户暂停、预算耗尽）。pi 的 follow-up 队列（第 8 课）需要调用方主动续上；Codex 把“续上”做进了运行时。本课只设置目标、不发任何消息，看 Codex 自己跑三个回合。约 45 分钟。

fn on_thread_idle<'a>(&'a self, input: ThreadIdleInput<'a>) -> ExtensionFuture<'a, ()> {
    Box::pin(async move {
        let Some(runtime) = goal_runtime_handle(input.thread_store) else {
            return;
        };

        let accounting = runtime.accounting_state();
        if let Some(turn_id) = accounting.interrupted_turn_id() {
            if let Err(err) = runtime
                .stop_active_goal_for_turn(&turn_id, ActiveGoalStopReason::EmptyResponse)
                .await
            {
                tracing::warn!("failed to stop goal after interrupted empty turns: {err}");
            }
            accounting.finish_turn(&turn_id);
        }
        if let Err(err) = runtime.continue_if_idle().await {
            tracing::warn!(
                "failed to continue active goal for idle thread {}: {err}",
                runtime.thread_id()
            );
        }
    })

cat > codex/scratch/s_goal.py <<'EOF'
# 剧本：每次被“继续目标”提示唤醒就做一步；第 3 步完成后调用 update_goal(complete)
import json
step = 0
def reply(n, body):
    global step
    last = body["input"][-1]
    if last.get("type") == "function_call_output":
        return [msg(f"第 {step} 步完成")] if step < 3 else [msg("目标达成，已标记 complete")]
    texts = [c.get("text", "") for i in body["input"] if i.get("type") == "message" for c in i["content"]]
    cont = sum("Continue working toward the active thread goal" in t for t in texts)
    step += 1
    print(f"[mock] req{n:02d} 历史中的继续提示 {cont} 条 → 第 {step} 步", flush=True)
    if step < 3: return [call("exec_command", {"cmd": f"echo step{step} >> log.txt"}, f"s{step}")]
    return [call("update_goal", {"status": "complete"}, "done")]
EOF

# /goal：设置目标后，线程空闲时 Codex 自己开启“继续”回合，直到模型调用 update_goal(complete)
import os, sys, time; sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from apc import AppServer
a = AppServer(cwd=os.getcwd()); tid = a.start(os.getcwd())
g = a.request("thread/goal/set", {"threadId": tid, "objective": "把 step1..step3 写进 log.txt", "origin": "user"})["goal"]
print("thread/goal/set →", {k: g[k] for k in ("status", "tokensUsed", "tokenBudget")})
turns = 0
def show(m):
    global turns
    p = m.get("params", {})
    if m["method"] == "turn/started": turns += 1; print(f"  turn/started #{turns}（没有任何用户输入）")
    if m["method"] == "thread/goal/updated": print("  thread/goal/updated → status =", p["goal"]["status"])
    if m["method"] == "item/completed" and p["item"]["type"] == "agentMessage": print("   ", p["item"]["text"])
while True:
    p = a.wait("thread/goal/updated", show=show, timeout=60)
    if p["goal"]["status"] != "active": break
time.sleep(1)
print("thread/goal/get →", a.request("thread/goal/get", {"threadId": tid})["goal"]["status"])
print("log.txt:", open("log.txt").read().split())
a.close()

在 thread/goal/set 中加入 "tokenBudget": 2500（mock 每次报告 1000×n 个输入 token）。会在第几步停下？状态是什么？模型最后收到的是哪个模板？
查看参考答案
作者实测：第 1 个继续回合有两次请求，mock 报告 1000+2000 = 3000 输入 token，超过 2500；回合结束时状态变为 budgetLimited（thread/goal/updated 与 thread/goal/get 都能看到），之后不再自动续跑，log.txt 中只有 step1。这次没有出现第三个请求：budget_limit.md 模板（budget_limit_steering_item）只在工具执行结束时注入（extension.rs L513–550：每次工具完成后结算 token，发现 BudgetLimited 就用 inject_active_turn_steering 把模板塞进进行中的回合，让模型收尾汇报）；工具 s1 结束时只计入了 1000 token，本例中越界发生在回合的最后一个请求，回合已经结束，所以没有机会注入。

小结
记住这几点


  1. 特性：Feature::Goals 稳定且默认开启（features/src/lib.rs）。实现在扩展 crate ext/goal（约 3400 行）。
  2. 工具（spec.rs）：get_goal、create_goal（只在用户明确要求时）、update_goal（status 只能是 complete/blocked/paused；blocked 要求同一阻塞连续出现至少 3 个目标回合；paused 只能应用户要求）。这些规则写在工具描述里，直接给模型看。
  3. 存储：目标存在 CODEX_HOME 下的 SQLite（goals_1.sqlite），所以恢复线程后目标还在（restore_after_resume）。
  4. 续跑循环：extension.rs 的 on_thread_idle → runtime.rs 的 continue_if_idle：工具不可用、有延迟标记、目标不是 Active 都直接返回；否则用 continuation_steering_item 构造提示，start_turn_if_idle 以 turn_trigger="goal" 开启回合。
  5. 继续提示（templates/goals/continuation.md）：把 objective 当作数据放进 <objective>（XML 转义），附上已用/预算/剩余 token，并要求“以当前工作区为准、保持完整目标”。未启用 update_plan 时使用去掉相关说明的版本。另有预算耗尽与目标被修改两种模板。
  6. 入口：TUI /goal 目标文本（还接受 /goooal）；app-server thread/goal/set|get|clear 与通知 thread/goal/updated|cleared。
