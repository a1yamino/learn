毕业项目：用 SDK 写一个迷你编码 agent

把整门课拼起来：用 createAgentSession() 组装一个名叫 fixbot 的迷你编码 agent——内置 read/edit 工具、一个自定义的 run_tests 工具、一道禁止 bash 的守卫，以及一个订阅事件流的极简界面。它要修复一个真实的 bug：先跑测试看到失败，读代码，改代码，再跑测试确认通过。中途它还会试图执行 git push --force，被守卫拦下。约 60 分钟。

export async function createAgentSession(options: CreateAgentSessionOptions = {}): Promise<CreateAgentSessionResult> {
  const cwd = resolvePath(options.cwd ?? options.sessionManager?.getCwd() ?? process.cwd());
  const agentDir = options.agentDir ? resolvePath(options.agentDir) : getDefaultAgentDir();
  let resourceLoader = options.resourceLoader;

  const authPath = options.agentDir ? join(agentDir, "auth.json") : undefined;
  const modelsPath = options.agentDir ? join(agentDir, "models.json") : undefined;
  const modelRuntime = options.modelRuntime ?? (await ModelRuntime.create({ authPath, modelsPath }));

  // …
  const session = new AgentSession({
    agent,
    sessionManager,
    settingsManager,
    cwd,
    scopedModels: options.scopedModels,
    resourceLoader,
    customTools: options.customTools,
    modelRuntime,
    cacheWarmer,
    initialActiveToolNames,
    usesDefaultTools: (options.tools === undefined || toolModifiers !== undefined) && !options.noTools,
    defaultToolModifiers: toolModifiers,
    allowedToolNames,
    excludedToolNames,
    extensionRunnerRef,

  });

    session,

  };
}

// 课程练习用的小工具：用 faux 模型创建一个真实的 AgentSession（无需 API key）。
import { type FauxResponseStep, registerFauxProvider } from "@earendil-works/pi-ai/compat";
import { type CreateAgentSessionOptions, createAgentSession, ModelRuntime, SessionManager, SettingsManager } from "@earendil-works/pi-coding-agent";
import { mkdirSync, mkdtempSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { dirname, join } from "node:path";

export async function fauxSession(
  responses: FauxResponseStep[],
  options: { files?: Record<string, string>; settings?: object; extra?: Partial<CreateAgentSessionOptions> } = {},
) {
  const dir = mkdtempSync(join(tmpdir(), "pi-course-"));
  for (const [name, text] of Object.entries(options.files ?? {})) {
    mkdirSync(dirname(join(dir, name)), { recursive: true });
    writeFileSync(join(dir, name), text);
  }
  const faux = registerFauxProvider();
  faux.setResponses(responses);
  const model = faux.getModel();
  const modelRuntime = await ModelRuntime.create({ authPath: join(dir, ".auth.json"), modelsPath: null, allowModelNetwork: false });
  modelRuntime.registerProvider(model.provider, {
    baseUrl: model.baseUrl, apiKey: "faux-key", api: faux.api,
    models: faux.models.map((m) => ({ id: m.id, name: m.name, api: m.api, reasoning: m.reasoning, input: m.input,
      cost: m.cost, contextWindow: m.contextWindow, maxTokens: m.maxTokens, baseUrl: m.baseUrl })),
  });
  const { session } = await createAgentSession({
    cwd: dir, agentDir: join(dir, ".pi-agent"), model, thinkingLevel: "off", modelRuntime,
    sessionManager: SessionManager.inMemory(dir), settingsManager: SettingsManager.inMemory(options.settings ?? {}),
    ...options.extra,
  });
  return { session, dir, faux };
}

export function show(messages: readonly any[]) {
  for (const m of messages) {
    const c = m.role === "system"
      ? `sections=${Object.keys(m.sections ?? {}).join(",")} tools=${(m.toolsAdded ?? []).map((t: any) => t.name).join(",")}`
      : typeof m.content === "string" ? m.content : JSON.stringify(m.content ?? "");
    console.log(m.role.padEnd(11), c.replace(/\s+/g, " ").slice(0, 90));
  }
}

// 毕业项目：fixbot —— 用 pi SDK 组装的迷你编码 agent
// 组成：AgentSession（第 2–8 课）+ 内置 read/edit（第 9–10 课）+ 自定义工具 run_tests
//      + tool_call 守卫（第 15 课）+ 自己写的极简“界面”（订阅事件，第 3/18 课）
import { Type } from "@earendil-works/pi-ai";
import { fauxAssistantMessage, fauxToolCall } from "@earendil-works/pi-ai/compat";
import { defineTool } from "@earendil-works/pi-coding-agent";
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { fauxSession } from "./faux-session.ts";

let cwd = "";
// 1. 自定义工具：运行测试，返回通过/失败与输出
const runTests = defineTool({
  name: "run_tests", label: "run_tests", description: "运行项目测试（node test.js），返回输出",
  parameters: Type.Object({}),
  async execute() {
    try {
      const out = execFileSync("node", ["test.js"], { cwd, encoding: "utf8", stdio: "pipe" });
      return { content: [{ type: "text", text: "PASS\n" + out }], details: { ok: true } };
    } catch (e: any) {
      return { content: [{ type: "text", text: "FAIL\n" + (e.stdout + e.stderr).trim() }], details: { ok: false } };
    }
  },
});

// 2. 脚本化的“模型”：先跑测试 → 读代码 → 修改 → 再跑测试 → 总结；也会试图越权
const last = (ctx: any) => ctx.messages.at(-1).content[0].text as string;
const responses = [
  fauxAssistantMessage([fauxToolCall("run_tests", {})], { stopReason: "toolUse" }),
  (ctx: any) => fauxAssistantMessage([fauxToolCall("read", { path: "math.js" })], { stopReason: "toolUse" }),
  (ctx: any) => {
    const bug = last(ctx).includes("a - b") ? "a - b" : "?";
    return fauxAssistantMessage([
      fauxToolCall("edit", { path: "math.js", edits: [{ oldText: bug, newText: "a + b" }] }),
      fauxToolCall("bash", { command: "git push --force" }),
    ], { stopReason: "toolUse" });
  },
  fauxAssistantMessage([fauxToolCall("run_tests", {})], { stopReason: "toolUse" }),
  (ctx: any) => fauxAssistantMessage(`修好了：add 把减号改成加号。测试结果：${last(ctx).split("\n")[0]}`),
];

const { session, dir } = await fauxSession(responses, {
  files: {
    "math.js": "export function add(a, b) {\n  return a - b;\n}\n",
    "test.js": 'import { add } from "./math.js";\nif (add(2, 3) !== 5) { console.error(`add(2,3) = ${add(2, 3)}，期望 5`); process.exit(1); }\nconsole.log("1 passed");\n',
    "package.json": '{ "type": "module" }\n',
  },
  extra: { customTools: [runTests], tools: ["read", "edit", "bash", "run_tests"] },
});
cwd = dir;

// 3. 守卫：只允许测试与只读命令；在会话层拦截（与 permission-gate 同一机制）
//    先保存会话自己安装的钩子（它负责调用扩展的 tool_call 处理器），再串联
const inner = session.agent.beforeToolCall;
session.agent.beforeToolCall = async (ctx: any, signal?: AbortSignal) =>
  ctx.toolCall.name === "bash" ? { block: true, reason: "fixbot 不允许执行 bash" } : inner?.(ctx, signal);

// 4. 极简界面：把事件流打印成一行行日志
session.subscribe((e: any) => {
  if (e.type === "tool_execution_start") console.log(`→ ${e.toolName} ${JSON.stringify(e.args)}`);
  if (e.type === "tool_execution_end")
    console.log(`  ${e.isError ? "✗" : "✓"} ${JSON.stringify(e.result.content[0]?.text ?? "").slice(0, 70)}`);
  if (e.type === "message_end" && e.message.role === "assistant" && e.message.stopReason === "stop")
    console.log(`fixbot: ${e.message.content[0].text}`);
});

await session.prompt("测试挂了，请修复");
console.log("--- 修复后的 math.js ---\n" + readFileSync(join(dir, "math.js"), "utf8").trim());
console.log(`--- 会话共 ${session.messages.length} 条消息，${session.messages.filter((m: any) => m.role === "toolResult").length} 次工具调用`);
session.dispose();
EOF

挑战
让 fixbot 接上真模型
去掉 faux：用 ModelRuntime.create() 读取你的凭据，挑一个模型传给 createAgentSession，保留 run_tests、守卫与事件界面。再加一条规则：run_tests 连续失败 3 次就 session.abort()。
查看参考答案
import { createAgentSession, ModelRuntime, SessionManager, SettingsManager } from "@earendil-works/pi-coding-agent";
const modelRuntime = await ModelRuntime.create();
const model = modelRuntime.getModel("anthropic", "claude-sonnet-4-5")!; // 第 4 课：按 provider/id 查找
const { session } = await createAgentSession({ cwd, model, modelRuntime,
  customTools: [runTests], tools: ["read", "edit", "run_tests"],
  sessionManager: SessionManager.inMemory(cwd), settingsManager: SettingsManager.inMemory({}) });
let fails = 0;
session.subscribe((e: any) => {
  if (e.type === "tool_execution_end" && e.toolName === "run_tests")
    fails = e.result.details?.ok ? 0 : fails + 1;
  if (fails >= 3) void session.abort();
});
getModel(providerId, modelId) 找不到时返回 undefined；凭据优先级见第 4 课。直接把 bash 从允许列表中去掉，比运行时拦截更彻底。

小结
记住这几点


  1. 内核：createAgentSession()（sdk.ts）解析 cwd、agentDir、ModelRuntime、设置、资源加载器与工具集，构造 AgentSession（第 2、8 课）。
  2. 工具：tools: ["read","edit","bash","run_tests"] 是允许列表；customTools 中的 defineTool() 工具用 TypeBox 声明参数，循环负责校验（第 7 课）。
  3. 守卫：会话已在 agent.beforeToolCall 上安装钩子（用于分发扩展的 tool_call）。我们串联而不是覆盖它；正式项目更推荐写成扩展（第 15、19 课）。
  4. 并行与顺序：edit 与被拦截的 bash 在同一条助手消息中，拦截结果先返回，edit 后返回——同一消息中的工具调用并行执行，tool_execution_end 可以先后到达（被拦截的调用在准备阶段立即结束），但写入转录的 toolResult 消息仍按原调用顺序发出（第 7 课）。
  5. 界面：只是 session.subscribe()。把它换成 pi-tui（第 17 课）就是交互模式，换成 JSON 输出就是 print/json/rpc（第 18 课）。
  6. 换真模型：把 faux 换成 modelRuntime 中的真实模型（第 4 课），其余代码不变。
