class PendingMessageQueue {
  private messages: AgentMessage[] = [];
  public mode: QueueMode;

  constructor(mode: QueueMode) {
    this.mode = mode;
  }

  enqueue(message: AgentMessage): void {
    this.messages.push(message);
  }

  hasItems(): boolean {
    return this.messages.length > 0;
  }

  peek(): AgentMessage[] {
    if (this.mode === "all") return this.messages.slice();
    const first = this.messages[0];
    return first ? [first] : [];
  }

  drain(): AgentMessage[] {
    const drained = this.peek();
    this.messages = this.messages.slice(drained.length);
    return drained;
  }

  clear(): void {
    this.messages = [];
  }

private async runWithLifecycle(executor: (signal: AbortSignal) => Promise<void>): Promise<void> {
  if (this.activeRun) {
    throw new Error("Agent is already processing.");
  }

  const abortController = new AbortController();
  let resolvePromise = () => {};
  const promise = new Promise<void>((resolve) => {
    resolvePromise = resolve;
  });
  this.activeRun = { promise, resolve: resolvePromise, abortController };

  this._state.isStreaming = true;
  this._state.streamingMessage = undefined;
  this._state.errorMessage = undefined;

  try {
    await executor(abortController.signal);
  } catch (error) {
    await this.handleRunFailure(error, abortController.signal.aborted);
  } finally {
    this.finishRun();
  }
}
事件归约：
packages/agent/src/agent.ts  L575–596, L614–622 ↗
private async processEvents(event: AgentEvent): Promise<void> {
  switch (event.type) {
    case "message_start":
      this._state.streamingMessage = event.message;
      break;

    case "message_update":
      this._state.streamingMessage = event.message;
      break;

    case "message_end":
      this._state.streamingMessage = undefined;
      this._state.messages.push(event.message);
      break;

    case "tool_execution_start": {
      const pendingToolCalls = new Set(this._state.pendingToolCalls);
      pendingToolCalls.add(event.toolCallId);
      this._state.pendingToolCalls = pendingToolCalls;
      break;
    }

  // …

  const signal = this.activeRun?.abortController.signal;
  if (!signal) {
    throw new Error("Agent listener invoked outside active run");
  }
  for (const listener of this.listeners) {
    await listener(event, signal);
  }
}

import { Type } from "@earendil-works/pi-ai";
import { fauxAssistantMessage, fauxToolCall, registerFauxProvider, streamSimple } from "@earendil-works/pi-ai/compat";
import { Agent } from "@earendil-works/pi-agent-core";

const faux = registerFauxProvider();
const wait = {
  name: "wait", label: "wait", description: "等待", parameters: Type.Object({}),
  execute: async (_id: string, _p: {}, signal?: AbortSignal) => {
    await new Promise<void>((r, j) => { const t = setTimeout(r, 200); signal?.addEventListener("abort", () => { clearTimeout(t); j(new Error("被中止")); }); });
    return { content: [{ type: "text" as const, text: "done" }], details: {} };
  },
};
const user = (t: string) => ({ role: "user" as const, content: t, timestamp: Date.now() });
const text = (m: any) => (typeof m.content === "string" ? m.content : m.content.map((c: any) => c.text ?? `<${c.type}:${c.name}>`).join(""));

function makeAgent(mode: "one-at-a-time" | "all") {
  faux.setResponses([fauxAssistantMessage([fauxToolCall("wait", {})], { stopReason: "toolUse" }), ...Array.from({ length: 4 }, (_, i) => fauxAssistantMessage(`回复${i + 1}`))]);
  return new Agent({ streamFn: streamSimple, steeringMode: mode,
    initialState: { systemPrompt: "", model: faux.getModel(), thinkingLevel: "off", tools: [wait] } });
}

for (const mode of ["one-at-a-time", "all"] as const) {
  const agent = makeAgent(mode);
  agent.subscribe((e) => { if (e.type === "tool_execution_start") { agent.steer(user("插话A")); agent.steer(user("插话B")); } });
  await agent.prompt("开始");
  console.log(`== steeringMode=${mode}`);
  console.log("  ", agent.state.messages.filter((m) => m.role !== "system").map((m) => `${m.role}:${text(m)}`).join(" | "));
}

const agent = makeAgent("all");
agent.subscribe((e) => { if (e.type === "tool_execution_start") setTimeout(() => agent.abort(), 50); });
const run = agent.prompt("开始");
try { await agent.prompt("再来一个"); } catch (e) { console.log("== 运行中再次 prompt:", (e as Error).message.slice(0, 60)); }
await run;
const last = agent.state.messages.at(-1) as any;
console.log("== abort 后: 最后一条", last.role, last.stopReason ?? "", "| 工具结果:", text(agent.state.messages.find((m) => m.role === "toolResult")));
console.log("   isStreaming =", agent.state.isStreaming);

挑战
慢订阅者
给 agent 添加一个订阅者：在 message_end 时 await 300ms（模拟写磁盘）。总运行时间如何变化？如果改成不 await（fire-and-forget），会有什么风险？
查看参考答案
processEvents 会依次 await 每个订阅者，所以每条完成的消息都会让循环多等 300ms：本例中约 5 条消息 → 运行时间增加约 1.5 秒。这是有意的设计：AgentSession 的持久化就在 message_end 订阅者里，保证消息按顺序、在下一步开始前写入。
如果不 await，循环不会等待，但可能出现：进程在写入完成前退出导致会话丢失消息；多次写入交错导致顺序错乱；waitForIdle() resolve 时数据其实还没落盘。真正需要慢操作又不想阻塞时，应在订阅者内部自己排队并在合适时机刷新。

小结
记住这几点


  1. AgentState：systemPrompt、model、thinkingLevel、tools、messages，以及运行时字段 isStreaming、streamingMessage（正在流入的消息）、pendingToolCalls、errorMessage。
  2. processEvents：每个循环事件先更新状态（message_end 时把消息加入 messages；tool_execution_start/end 维护 pendingToolCalls），再依次 await 每个订阅者。订阅者变慢，循环就等待——这保证了持久化等副作用的顺序。
  3. 两个队列：steer() 进入 steering 队列，在当前 turn 的工具执行完后、下一次模型请求前注入；followUp() 进入 follow-up 队列，只在 agent 本来要停下时注入。正在执行的工具不会被跳过。
  4. 队列模式：one-at-a-time（默认）每次只取一条，两条插话按两次队列读取分别注入；每次读取后都会产生新的模型请求，但期间可能还有工具调用；all 一次取出全部。
  5. 单一活动运行：运行中调用 prompt() 会抛错并提示使用 steer/followUp；continue() 从现有转录继续（最后一条是 assistant 时，先消费队列）。
  6. 中止：abort() 触发当前运行的 AbortSignal；工具与提供方都收到它。运行中的未预期异常被转换为一条 error/aborted 的 assistant 消息，事件序列保持完整。waitForIdle() 在 agent_end 的订阅者全部完成后才 resolve。
