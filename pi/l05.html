function mapStopReason(reason: ChatCompletionChunk.Choice["finish_reason"] | string): {
  stopReason: StopReason;
  errorMessage?: string;
} {
  if (reason === null) return { stopReason: "stop" };
  switch (reason) {
    case "stop":
    case "end":
      return { stopReason: "stop" };
    case "length":
      return { stopReason: "length" };
    case "function_call":
    case "tool_calls":
      return { stopReason: "toolUse" };
    case "content_filter":
      return { stopReason: "error", errorMessage: "Provider finish_reason: content_filter" };
    case "network_error":
      return { stopReason: "error", errorMessage: "Provider finish_reason: network_error" };
    default:
      return {
        stopReason: "error",
        errorMessage: `Provider finish_reason: ${reason}`,
      };
  }
}

小结
记住这几点


  1. 请求方向：transformMessages（所有协议共用）先做跨模型清理，再由各协议的 convertMessages 转成线上格式（如 Chat Completions 的 messages、tools）。
  2. 跨模型切换：同一会话中途换模型时，transformMessages 丢弃其他模型的 thinking 块（签名与加密推理只对产生它的模型有效，明文思考又会被新模型误当作可见回复），去掉 Google 的 thoughtSignature，并可按目标 API 规范化工具调用 id。
  3. 孤儿工具调用：如果某个工具调用在转录里没有结果（例如被中断），会插入一个合成的 toolResult：“No result provided”，isError=true——许多 API 拒绝不配对的工具调用。
  4. 响应方向：逐块读取 SSE；reasoning_content / reasoning / reasoning_text → thinking；content → text；tool_calls 按 index 累积参数片段，用 parseStreamingJson 随时给出“到目前为止”的参数对象（UI 可以在参数还没流完时就显示路径）。
  5. 停止原因映射：stop/end→stop、length→length、tool_calls/function_call→toolUse，content_filter 等→error 并带说明。原始值保存在 rawStopReason。没有 finish_reason 的流被视为错误（除非该兼容端点声明不支持 finish_reason）。
  6. 错误永不抛出：任何异常都被捕获，转成一条 stopReason=error（或 aborted）的 assistant 消息，并作为 error 事件推送。已流出的内容保留。
  7. 兼容性开关：OpenAI 兼容服务各有怪癖，detectCompat 按 provider 与 baseUrl 自动探测（是否支持 developer 角色、usage 放在哪、思考字段格式……），模型的 compat 字段可以覆盖。
